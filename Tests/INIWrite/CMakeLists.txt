cmake_minimum_required(VERSION 3.14)
project(INIWrite LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The INI write failure tests require Linux and /dev/full")
endif()
find_package(Threads REQUIRED)
add_executable(INIWrite main.cpp ../../WDL/swell/swell-ini.cpp)
target_compile_features(INIWrite PRIVATE cxx_std_17)
target_link_libraries(INIWrite PRIVATE Threads::Threads)
enable_testing()
add_test(NAME INIWrite COMMAND INIWrite)
